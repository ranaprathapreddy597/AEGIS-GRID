cmake_minimum_required(VERSION 3.25)
project(aegis_twin VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(nlohmann_json 3.11 REQUIRED)
add_library(twin src/twin.cpp)
set_target_properties(twin PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(twin PUBLIC nlohmann_json::nlohmann_json)
target_include_directories(twin PUBLIC include)
if(MSVC)
  target_compile_options(twin PRIVATE /W4 /WX)
else()
  target_compile_options(twin PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(aegis-twin src/main.cpp)
target_link_libraries(aegis-twin PRIVATE twin)
add_library(aegis_twin SHARED src/twin_c.cpp)
target_link_libraries(aegis_twin PRIVATE twin)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  target_link_options(aegis-twin PRIVATE -static-libstdc++ -static-libgcc)
  target_link_options(aegis_twin PRIVATE -static-libstdc++ -static-libgcc)
endif()
if(BUILD_TESTING)
  add_executable(twin-tests tests/twin_tests.cpp)
  target_link_libraries(twin-tests PRIVATE twin)
  add_test(NAME thermal_kernel COMMAND twin-tests)
  add_test(NAME cli_rejects_bad_seed COMMAND aegis-twin --seed -1)
  set_tests_properties(cli_rejects_bad_seed PROPERTIES WILL_FAIL TRUE)
  add_test(NAME cli_rejects_zero_dt COMMAND aegis-twin --dt 0)
  set_tests_properties(cli_rejects_zero_dt PROPERTIES WILL_FAIL TRUE)
endif()
